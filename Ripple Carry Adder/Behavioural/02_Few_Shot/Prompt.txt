Here is a working 4-bit ripple carry adder using a for loop, 
as a style reference: 
 
// --- EXAMPLE (4-bit, do NOT copy, use as style guide) --- 
module rca_4bit(input [3:0] a, b, input cin, 
                 output reg [3:0] sum, output reg cout); 
  reg [4:0] carry; 
  integer i; 
  always @(*) begin 
    carry[0] = cin; 
    for (i = 0; i < 4; i = i + 1) begin 
      sum[i]     = a[i] ^ b[i] ^ carry[i]; 
      carry[i+1] = (a[i] & b[i]) | (b[i] & carry[i]) | (a[i] & carry[i]); 
    end 
    cout = carry[4]; 

  end 
endmodule 
// --- END EXAMPLE ---
